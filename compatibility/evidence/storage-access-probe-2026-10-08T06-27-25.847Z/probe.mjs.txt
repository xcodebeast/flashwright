import assert from 'node:assert/strict';
import { once } from 'node:events';
import { createServer } from 'node:http';
import { mkdir, writeFile, readFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { chromium } from 'playwright';
const reportDirectory = `compatibility/evidence/storage-access-probe-${new Date().toISOString().replaceAll(':', '-')}`;
await mkdir(reportDirectory);
const requests = [];
const server = createServer((request, response) => { requests.push(request.url); response.setHeader('content-type', 'text/html'); response.end('<title>Storage access</title>'); });
server.listen(0, '127.0.0.1');
await once(server, 'listening');
const origin = `http://127.0.0.1:${server.address().port}`;
const report = { scope: 'Exploratory real-browser CDP behavior, not Flashwright production or performance evidence.', browserPath: process.env.FLASHWRIGHT_BROWSER_PATH, records: [], requests };
let browser;
try {
  browser = await chromium.launch({ executablePath: report.browserPath });
  const context = await browser.newContext();
  const page = await context.newPage();
  const session = await context.newCDPSession(page);
  const storageId = { securityOrigin: origin, isLocalStorage: true };
  async function command(method, parameters, label) {
    try { const result = await session.send(method, parameters); report.records.push({ label, method, result }); return result; }
    catch (error) { report.records.push({ label, method, error: error.message }); return null; }
  }
  await command('DOMStorage.enable', {}, 'enable');
  await command('DOMStorage.setDOMStorageItem', { storageId, key: 'probe', value: 'before navigation' }, 'restore from about blank');
  await command('DOMStorage.getDOMStorageItems', { storageId }, 'read from about blank');
  await page.goto(origin);
  await page.evaluate(() => localStorage.setItem('probe', 'visited'));
  await command('DOMStorage.getDOMStorageItems', { storageId }, 'read visited origin');
  await page.goto('about:blank');
  await command('DOMStorage.getDOMStorageItems', { storageId }, 'read after leaving origin');
  await context.addCookies([{ name: 'remove', value: 'one', url: origin }, { name: 'keep', value: 'two', url: origin }]);
  const browserSession = await browser.newBrowserCDPSession();
  const { browserContextIds } = await browserSession.send('Target.getBrowserContexts');
  assert.equal(browserContextIds.length, 1);
  const { cookies } = await browserSession.send('Storage.getCookies', { browserContextId: browserContextIds[0] });
  const expired = cookies.filter(cookie => cookie.name === 'remove').map(({ name, value, domain, path, httpOnly, secure, sameSite }) => ({ name, value, domain, path, httpOnly, secure, sameSite, expires: 0 }));
  await browserSession.send('Storage.setCookies', { browserContextId: browserContextIds[0], cookies: expired });
  const remaining = await context.cookies();
  assert.deepEqual(remaining.map(cookie => cookie.name), ['keep']);
  report.records.push({ label: 'selective expiration preserves unrelated cookie', cookies: remaining });
  await context.close();
} finally {
  await browser?.close();
  await new Promise(resolve => server.close(resolve));
  report.scriptSha256 = createHash('sha256').update(await readFile(import.meta.filename)).digest('hex');
  report.browserSha256 = createHash('sha256').update(await readFile(report.browserPath)).digest('hex');
  report.finished = new Date().toISOString();
  await writeFile(`${reportDirectory}/probe.mjs.txt`, await readFile(import.meta.filename), { flag: 'wx' });
  await writeFile(`${reportDirectory}/report.json`, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ reportDirectory, records: report.records }, null, 2));
}
