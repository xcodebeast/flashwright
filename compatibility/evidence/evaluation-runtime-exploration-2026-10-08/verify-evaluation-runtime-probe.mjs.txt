import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile, readdir, writeFile } from 'node:fs/promises';
import { join, resolve } from 'node:path';
import { createValueCodec } from '../packages/resident-runtime/serialization.cjs';

const directory = resolve(process.argv[2]);
const read = async name => JSON.parse(await readFile(join(directory,name),'utf8'));
const digest = value => createHash('sha256').update(value).digest('hex');
const metadata = await read('metadata.json');
const report = await read('report.json');
assert.equal(report.status,'passed');
assert.deepEqual(report.sources,metadata.sources);
assert.deepEqual(report.configuration,metadata.configuration);
assert.equal(digest(await readFile(join(directory,'probe.mjs.txt'))),metadata.probeSha256);
for (const [path,hash] of Object.entries(metadata.sources)) assert.equal(digest(await readFile(path)),hash,path);
assert.equal(digest(await readFile(metadata.browser.executablePath)),metadata.browser.sha256);
const {configuration} = metadata;
assert.equal(configuration.independentLaunches,4);
assert.deepEqual(configuration.variants,['repeated-source','private-function']);
assert.ok(configuration.cases.includes('scalar') && configuration.cases.includes('structured'));
assert.equal(report.records.length,8);
const codec = createValueCodec();
let verifiedPayloads = 0;
const loaders = new Set();
for (let launch = 0; launch < configuration.independentLaunches; launch++) {
 const expectedOrder = launch % 2 ? [...configuration.variants].reverse() : configuration.variants;
 const records = report.records.filter(record => record.launch === launch);
 assert.deepEqual(records.map(record => record.variant),expectedOrder);
 for (const record of records) {
  assert.deepEqual(await read(`launch-${launch}-${record.variant}.json`),record);
  assert.ok(record.loader && !loaders.has(record.loader)); loaders.add(record.loader);
  const expectedCount = (configuration.warmupIterations + configuration.measuredIterations) * configuration.cases.length;
  assert.equal(record.calls.length,expectedCount);
  assert.equal(record.completedEvaluations,expectedCount);
  assert.ok(Number.isFinite(record.preparationMilliseconds) && record.preparationMilliseconds >= 0);
  assert.ok(record.completeMilliseconds >= record.preparationMilliseconds + record.calls.reduce((total,call) => total + call.milliseconds,0));
  assert.equal(record.preparationParameterBytes > 0,record.variant === 'private-function');
  for (const [index,call] of record.calls.entries()) {
   const iteration = Math.floor(index / configuration.cases.length);
   assert.equal(call.iteration,iteration);
   assert.equal(call.name,configuration.cases[index % configuration.cases.length]);
   assert.equal(call.phase,iteration < configuration.warmupIterations ? 'warmup' : 'measured');
   assert.equal(call.method,record.variant === 'private-function' ? 'Runtime.callFunctionOn' : 'Runtime.evaluate');
   assert.ok(Number.isFinite(call.milliseconds) && call.milliseconds > 0);
   assert.ok(Number.isSafeInteger(call.parameterBytes) && call.parameterBytes > 0);
   const success = ['scalar','structured','unicode-source'].includes(call.name);
   assert.equal(call.outcome,success ? 'success' : 'exception');
   if (success) { assert.equal(call.stage,undefined); assert.equal(call.errorCode,undefined); }
   else {
    assert.equal(call.stage,call.name === 'user-error' ? 'execution' : 'resultSerialization');
    assert.equal(call.errorCode,call.name === 'user-error' ? 'EvaluationError' : 'UnsupportedFeatureError');
   }
   if (call.result) {
    const argument = {iteration,text:'astral \u{1F680} lone \ud800',number:-0,date:new Date('2020-01-02T03:04:05Z'),expression:/abc/gi};
    argument.self = argument;
    assert.deepEqual(call.encodedArgument,codec.encode(argument));
    assert.equal(call.result.outcome,call.outcome);
    if (success) {
     const expected = call.name === 'scalar' ? 42 : call.name === 'unicode-source' ? 'astral \u{1F680} lone \ud800' : argument;
     assert.deepEqual(call.result.value,codec.encode(expected));
    } else {
     assert.equal(call.result.stage,call.stage);
     assert.equal(call.result.error.code,call.errorCode);
     assert.ok(call.result.error.messageUnits.every(unit => Number.isInteger(unit) && unit >= 0 && unit <= 65535));
     if (call.name === 'user-error') assert.ok(String.fromCharCode(...call.result.error.messageUnits).includes('expected probe failure'));
    }
    verifiedPayloads++;
   }
  }
 }
}
const totalCalls = report.records.reduce((total,record) => total + record.calls.length,0);
assert.ok(verifiedPayloads === 0 || verifiedPayloads === totalCalls);
if (configuration.cases.includes('unicode-source')) assert.equal(verifiedPayloads,totalCalls);
function median(values) {
 const sorted = [...values].sort((left,right) => left-right);
 return sorted.length % 2 ? sorted[Math.floor(sorted.length/2)] : (sorted[sorted.length/2-1]+sorted[sorted.length/2])/2;
}
const comparisons = {};
for (const name of configuration.cases) {
 const pairs = Array.from({length:4},(_,launch) => {
  const values = Object.fromEntries(configuration.variants.map(variant => {
   const samples = report.records.find(record => record.launch === launch && record.variant === variant).calls.filter(call => call.phase === 'measured' && call.name === name);
   assert.equal(samples.length,configuration.measuredIterations);
   return [variant,{medianMilliseconds:median(samples.map(sample => sample.milliseconds)),parameterBytes:median(samples.map(sample => sample.parameterBytes))}];
  }));
  return {launch,...values,referenceOverCandidate:values['repeated-source'].medianMilliseconds / values['private-function'].medianMilliseconds};
 });
 comparisons[name] = {pairs,medianPairedRatio:median(pairs.map(pair => pair.referenceOverCandidate))};
}
const completePairs = Array.from({length:4},(_,launch) => {
 const reference = report.records.find(record => record.launch === launch && record.variant === 'repeated-source');
 const candidate = report.records.find(record => record.launch === launch && record.variant === 'private-function');
 return {launch,referenceMilliseconds:reference.completeMilliseconds,candidateMilliseconds:candidate.completeMilliseconds,referenceOverCandidate:reference.completeMilliseconds / candidate.completeMilliseconds};
});
const files = (await readdir(directory)).filter(name => name !== 'verification.json');
const verification = {status:'passed',totalCalls,verifiedPayloads,independentLaunches:4,comparisons,completePairs,
 medianCompleteComponentRatio:median(completePairs.map(pair => pair.referenceOverCandidate)),
 retainedFileHashes:Object.fromEntries(await Promise.all(files.map(async name => [name,digest(await readFile(join(directory,name)))]))),
 scope:verifiedPayloads ? 'Full retained encoded arguments/results independently checked, with fresh document identities, complete application counters, source/browser hashes, ordering and recomputed component statistics. Four launch pairs are descriptive and do not establish production or suite benefit.' : 'Initial exploration retained live assertion checks but omitted encoded payloads. Only shape, scope, ordering, counters, source/browser hashes and component statistics can be independently checked; this is not the full-payload candidate evidence.'};
await writeFile(join(directory,'verification.json'),JSON.stringify(verification,null,2),{flag:'wx'});
console.log(JSON.stringify({directory,totalCalls,verifiedPayloads,medianCompleteComponentRatio:verification.medianCompleteComponentRatio,cases:Object.fromEntries(Object.entries(comparisons).map(([name,value]) => [name,value.medianPairedRatio]))}));
