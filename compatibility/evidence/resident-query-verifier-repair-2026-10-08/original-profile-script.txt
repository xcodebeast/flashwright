#!/usr/bin/env bash
set -euo pipefail
export PATH="$PWD/.tools/node-v24.15.0-darwin-arm64/bin:$PATH"
export FLASHWRIGHT_DRIVER_PATH="$PWD/target/release/flashwright-driver"
export FLASHWRIGHT_BROWSER_PATH='/Users/P/Library/Caches/ms-playwright/chromium_headless_shell-1223/chrome-headless-shell-mac-arm64/chrome-headless-shell'
export FLASHWRIGHT_RESIDENT_ASSERTIONS=1
export FLASHWRIGHT_CONDITION_INSTRUMENTATION=0
export FLASHWRIGHT_VERIFICATION_REPORT
export FLASHWRIGHT_BASELINE_VERIFICATION_REPORT
FLASHWRIGHT_VERIFICATION_REPORT=$(cat .cache/resident-core-query-batch-headless-on-path.txt)
FLASHWRIGHT_BASELINE_VERIFICATION_REPORT=$(cat .cache/resident-core-query-batch-headless-off-path.txt)
run_profile() {
 local profile="$1"
 local label="$2"
 local result_path
 npm run "benchmark:$profile" > ".cache/resident-query-$label.log" 2>&1
 result_path=$(python3 - "$label" <<'PY'
from pathlib import Path
import re,sys
label=sys.argv[1]
text=Path('.cache/resident-query-'+label+'.log').read_text()
path=re.findall(r'"destination": "([^"]+)"',text)[-1]
Path('.cache/resident-query-'+label+'-path.txt').write_text(path+'\n')
print(path)
PY
)
 node "tools/verify-$profile.mjs" "$result_path" > ".cache/resident-query-$label-verification.log" 2>&1
}
run_profile resident-load load
run_profile residency off-on
export FLASHWRIGHT_RESIDENT_ASSERTIONS=1
run_profile paired paired-first
run_profile paired paired-replication
